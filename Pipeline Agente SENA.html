Crear agente/executor.py — dispatcher HTTP

El executor traduce el nombre de cada herramienta a una llamada real al backend REST.

1
Crear agente/executor.py ▼
📄agente/executor.py
python 
import httpx, os

BACKEND = os.getenv("BACKEND_URL", "http://localhost:8000")

async def ejecutar(nombre: str, args: dict, token: str) -> dict:
    headers = {"Authorization": f"Bearer {token}"}
    async with httpx.AsyncClient() as c:
        if nombre == "listar_usuarios":
            r = await c.get(f"{BACKEND}/usuarios", headers=headers)
        elif nombre == "buscar_usuario":
            r = await c.get(f"{BACKEND}/usuarios",
                            params={"q": args["q"]}, headers=headers)
        else:
            return {"error": f"Herramienta desconocida: {nombre}"}
    return r.json()

Lab 2 · Paso 3 de 4

Crear agente/agent.py — el ciclo ReAct

Este es el núcleo del agente. El bucle for paso in range(6) implementa el ciclo Pensamiento → Acción → Observación.

1
Crear agente/agent.py ▼
📄agente/agent.py
python 
import asyncio, json, os
from openai import AsyncOpenAI
from dotenv import load_dotenv
from tools import TOOLS
from executor import ejecutar

load_dotenv()
client = AsyncOpenAI(
    api_key=os.getenv("API_KEY"),
    base_url=os.getenv("API_BASE")
)
MODEL = os.getenv("API_MODEL", "gemini-1.5-flash")

async def run(consulta: str, token: str):
    messages = [
        {"role": "system",
         "content": "Eres un asistente que gestiona usuarios SENA. "
                    "Responde siempre en español."},
        {"role": "user", "content": consulta}
    ]
    print(f"\n{'='*55}")
    print(f"Consulta: {consulta}")
    print('='*55)

    for paso in range(6):                   # max 6 ciclos ReAct
        resp = await client.chat.completions.create(
            model=MODEL, messages=messages, tools=TOOLS
        )
        msg = resp.choices[0].message

        # 1. PENSAMIENTO
        if msg.content:
            print(f"\n  💭 Pensamiento: {msg.content[:200]}")

        # 2. ¿Hay acción?
        if not msg.tool_calls:              # el LLM ya sabe responder
            print(f"\n  ✅ Respuesta: {msg.content}")
            break

        messages.append(msg)

        # 3. EJECUTAR herramienta → OBSERVACIÓN
        for tc in msg.tool_calls:
            args = json.loads(tc.function.arguments)
            print(f"\n  ⚡ Acción [{paso+1}]: {tc.function.name}({args})")
            resultado = await ejecutar(tc.function.name, args, token)
            obs = json.dumps(resultado, ensure_ascii=False)
            print(f"  📊 Observación: {obs[:300]}")
            messages.append({
                "role": "tool",
                "tool_call_id": tc.id,
                "content": json.dumps(resultado)
            })

  • Entiendo el ciclo: Pensamiento → Acción → Observación

import asyncio, httpx, os

from agent import run

Ampliar executor.py — todos los endpoints

Reemplaza agente/executor.py con la versión que cubre las 5 herramientas.

1
Reemplazar agente/executor.py (versión completa) ▼
python 
import httpx, os, json

BACKEND = os.getenv("BACKEND_URL", "http://localhost:8000")

async def ejecutar(nombre: str, args: dict, token: str) -> dict:
    h = {"Authorization": f"Bearer {token}"}
    async with httpx.AsyncClient() as c:
        if nombre == "listar_usuarios":
            r = await c.get(f"{BACKEND}/usuarios", headers=h)
        elif nombre == "buscar_usuario":
            r = await c.get(f"{BACKEND}/usuarios",
                            params={"q": args["q"]}, headers=h)
        elif nombre == "crear_usuario":
            r = await c.post(f"{BACKEND}/usuarios",
                             json=args, headers=h)
        elif nombre == "crear_proyecto":
            r = await c.post(f"{BACKEND}/proyectos",
                             json=args, headers=h)
        elif nombre == "asignar_usuario_a_proyecto":
            pid = args["proyecto_id"]
            uid = args["usuario_id"]
            r = await c.post(
                f"{BACKEND}/proyectos/{pid}/usuarios/{uid}",
                headers=h)
        else:
            return {"error": f"Herramienta no implementada: {nombre}"}
        try:
            return r.json()
        except Exception:
            return {"status_code": r.status_code, "texto": r.text}
